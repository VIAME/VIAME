# Requires-Dist, one per line. The part common to every CUDA variant; torch
# and the nvidia runtime live in requirements-cu<major>.txt.
#
# Established by running the GFIT and DEFAULT-FISH pipelines in an empty
# environment, not by scanning imports -- five entries below appear nowhere
# in VIAME's sources and were found only by a pipeline failing on them.
#
# Vendoring makes a fork's requirements ours: viame.rfdetr and viame.sam2 are
# inside this package. Where floors overlap, the highest wins.

# --- the interpreter and arrays -------------------------------------------
# numpy's floor is sam2's; ours was 1.13.
numpy>=1.24.4

# --- torch ----------------------------------------------------------------
# In the base requirements rather than behind an extra, by decision: nearly
# every pipeline that does anything needs it, and a wheel that installs but
# cannot run is the failure this file exists to prevent. It is also what
# brings the `nvidia-*` CUDA wheels the binaries' RUNPATH points at -- see
# `add_cuda_rpath` in build_wheel.py. Floors are sam2's, above rfdetr's
# 2.2.0 / 0.17.0.
# torch and the CUDA runtime are per-variant; see requirements-cu12.txt
# and requirements-cu13.txt.

# --- video -----------------------------------------------------------------
# PyAV backs `vidl_ffmpeg`, `ffmpeg` and `pyav`, which is how every shipped
# pipeline reads a video, and `viame/video_io/pyav_video_input.py` imports it
# directly. It is also where FFmpeg comes from: this wheel carries no libav,
# and PyAV's own wheel bundles the lot in `av.libs`.
#
# Declared although `supervision` happens to pull it in today. Depending on
# somebody else's dependency for the video reader is how video decoding stops
# working in a release nobody thought had touched it.
av

# The vendored fork trains through Lightning -- detr.py and every training
# callback import it at module scope -- and so does our own rf_detr_trainer.
# Without it `viame train` on any RF-DETR config dies at update_model with
# ModuleNotFoundError, after chipping the whole dataset.
pytorch_lightning>=2.0,<3
# The fork's COCO dataset and its evaluation callback both import this at
# module scope, so it is reached on every training run, not only when COCO
# input is given.
pycocotools
# torchmetrics' MeanAveragePrecision, which the fork's validation runs every
# epoch, is configured for the `faster_coco_eval` backend and raises at the
# first validation when it is absent. Nothing in our sources imports it, so
# an import scan does not find it -- only running a training does.
faster-coco-eval
# RF-DETR builds every augmentation through albumentations and silently drops
# the entire pipeline when the import fails, so a run configured for flips
# trains on unaugmented imagery instead. Not an import-time failure, which is
# why it is easy to miss.
albumentations>=2.0
