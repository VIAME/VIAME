# What goes into the viame wheel, on this branch.
#
#   include <glob relative to the install prefix> -> <destination in the wheel>
#   exclude <glob relative to the install prefix>
#
# `**` crosses directory separators, `*` does not, and excludes are applied
# after includes so a package can be taken whole and then thinned. `{data}`
# expands to `<name>-<version>.data/data`, the scheme pip unpacks into the
# environment prefix.
#
# Main ships VIAME and KWIVER, their native dependencies, and dynamic plugins.
# Lite has its own manifest for the merged library and single Python package.
include lib/python3.*/site-packages/viame/**  -> viame/
include lib/python3.*/site-packages/kwiver/** -> kwiver/

# Keep the install's shared-library dependencies and plugin discovery layout.
# Single-directory patterns preserve SONAMEs through the wheel builder.
include lib/*.so*          -> {data}/lib/
include lib/kwiver/**      -> {data}/lib/kwiver/
include lib/viame/**       -> {data}/lib/viame/

# ----------------------------------------------------------------------------
# The tools
#
# `viame` and `kwiver` are ELF executables, not python entry points, so they
# go through the wheel's `scripts` scheme: pip unpacks `.data/scripts/` into
# the environment's `bin/` and marks it executable, which puts `viame` on the
# PATH of an activated environment.
#
# They need no patching either. Their RUNPATH is `$ORIGIN/../lib:$ORIGIN/`,
# and from `<env>/bin` that first entry is `<env>/lib` -- the same directory
# `{data}/lib/` above unpacks into.
#
# The five shell scripts the manifest also installs under `bin/` are left
# out deliberately: they resolve their own paths against a `VIAME_INSTALL`
# layout with `configs/` and add-on archives beside them, which a code-only
# wheel does not have. Shipping them would put commands on the PATH that
# fail in a way that looks like a broken install rather than a missing
# component.
# ----------------------------------------------------------------------------
# Into `{data}/libexec/viame/` rather than onto the PATH, and a generated
# python launcher takes their place -- `--launcher` in wheel.cmake. These are
# ELF binaries linking `libpython3.X.so.1.0`, and nothing knows where that is
# until an interpreter is running: on a distribution python it is in the
# ldconfig cache, on a standalone one (pyenv, conda, uv, python.org) it is
# not, and the tool dies before main with "error while loading shared
# libraries". The launcher asks the running interpreter and sets
# LD_LIBRARY_PATH before exec'ing these.
# Flat under `libexec/`, not `libexec/viame/`: these carry
# `$ORIGIN/../lib` for libviame, and from `<prefix>/libexec/` that is
# `<prefix>/lib`, which is where `{data}/lib/` lands. One directory
# deeper and it resolves to `<prefix>/libexec/lib`, which is nothing.
include bin/viame   -> {data}/libexec/
include bin/kwiver  -> {data}/libexec/

# ----------------------------------------------------------------------------
# The tool scripts
#
# `viame run <name>`, `add-ons`, `index`, `search` and the rest are applets
# that shell out to a script in `configs/`. `python_script_applet` looks in
# `$VIAME_INSTALL/configs` and then `<exe>/../configs`; from `<env>/bin` the
# second is `<env>/configs`, which is what `{data}/` unpacks into. So these
# work with no environment set, and `run.py` finds the pipelines beside it
# because it resolves them from its own `__file__`.
# ----------------------------------------------------------------------------
# Add-on catalog: shared tool data, including in Windows Python environments.
include bin/download_viame_addons.csv -> {data}/configs/
include configs/*.py            -> {data}/configs/
include configs/gui-params/**   -> {data}/configs/gui-params/
include configs/prj-template/** -> {data}/configs/prj-template/

# ----------------------------------------------------------------------------
# What never ships
#
# `__pycache__` is the build tree's, compiled against a path the wheel will
# not have. Tests and their data are not what `pip install viame` is for.
# Static archives are link-time only.
#
# Not excluded here because they are not included above, and worth saying so:
# `configs/` is 8.6 GB of downloaded models, `include/` is for out-of-tree
# C++ plugin development, and `bin/` holds the tools. A code-only wheel is
# the decision on record; models are fetched separately.
# ----------------------------------------------------------------------------
exclude **/__pycache__/**
exclude **/*.pyc
exclude **/*.pyo
exclude **/tests/**
exclude **/test/**
exclude **/*.a
